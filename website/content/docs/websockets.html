WebSockets

Implement real-time bidirectional communication with WebSocket support in Suave.

Basic WebSocket Handler

Create a simple WebSocket echo server. Besides data frames, a handler should answer every Ping with a Pong carrying the same payload, and answer the client's Close with a Close before it stops reading:

open Suave
open Suave.WebSocket
open Suave.Sockets
open Suave.Sockets.Control
open System.Text
open Suave.Filters
open Suave.Operators
open Suave.RequestErrors

/// Pages allowed to open a socket; see "Check the Origin header" below.
let allowedOrigins = set [ "http://localhost:8080"; "https://example.com" ]

let checkOrigin : WebPart =
  fun ctx ->
    match ctx.request.header "origin" with
    | Choice1Of2 origin when allowedOrigins.Contains origin -> succeed ctx
    | _ -> FORBIDDEN "Origin not allowed" ctx

let echo (webSocket: WebSocket) (context: HttpContext) =
  socket {
    let mutable loop = true

    while loop do
      let! msg = webSocket.read()

      match msg with
      | (Text, data, true) ->
          let str = Encoding.UTF8.GetString data.Span
          let response = sprintf "Echo: %s" str
          let byteResponse =
            response
            |> Encoding.UTF8.GetBytes
            |> ByteSegment
          do! webSocket.send Text byteResponse true

      | (Ping, data, _) ->
          // Answer every Ping with a Pong carrying the same payload
          do! webSocket.send Pong data true

      | (Close, _, _) ->
          // Answer the client's Close with a Close, then stop reading
          do! webSocket.send Close (ByteSegment [||]) true
          loop <- false

      | _ -> ()
  }

let app =
  choose [
    path "/ws" >=> checkOrigin >=> handShake echo
  ]

Check the Origin header

Always check Origin before handShake. Browsers attach the user's cookies to a WebSocket handshake started by any page, and the same-origin policy and CORS do not apply to WebSockets, so without this check a malicious site can open a socket that is authenticated as your user and read the replies (cross-site WebSocket hijacking). The checkOrigin web part above answers 403 Forbidden unless the handshake comes from a page you list. Browsers always send Origin on WebSocket handshakes; refusing requests without it also turns away non-browser clients, so if you need those, authenticate them some other way (for example with a token) rather than by cookie.

Limits and failures

Suave refuses any frame whose announced payload exceeds SuaveConfig.webSocketMaxFrameSize (16 MiB by default): it sends a Close frame with code 1009 (message too big) and drops the connection before allocating a buffer, and read returns an error. Lower the limit to what your protocol needs, and raise it per socket with maxFrameSize where an endpoint needs more:

let config =
  { defaultConfig with
      webSocketMaxFrameSize = 64 * 1024 } // 64 KiB per frame

let upload (webSocket: WebSocket) (context: HttpContext) =
  // This endpoint takes larger frames than the rest of the server
  webSocket.maxFrameSize <- 8 * 1024 * 1024
  echo webSocket context

If the handler throws, Suave reports the exception through Globals.writeMessage, sends a Close frame with code 1011 (internal error) and closes the connection. Nothing is written as an HTTP response once the connection has switched protocols.

Broadcasting Messages

To fan messages out to many clients, keep the client table in a MailboxProcessor hub, check the result of every send, and drop clients whose send fails. The broadcast recipe has the complete hub. Any web part can then push a notification to every connected client:

let app =
  choose [
    path "/notifications" >=> checkOrigin >=> handShake ws
    POST >=> path "/notify" >=> fun ctx ->
      match ctx.request.formData "message" with
      | Choice1Of2 msg ->
          hub.Post (Broadcast msg)
          OK "Notification sent" ctx
      | _ -> BAD_REQUEST "Missing message" ctx
  ]

WebSocket with JSON Messages

Exchange structured JSON messages:

open Suave
open Suave.WebSocket
open System.Text
open System.Text.Json
open Suave.Filters
open Suave.Operators
open Suave.Sockets
open Suave.Sockets.Control

type Message = { user: string; text: string; timestamp: System.DateTime }

let jsonWebSocket (webSocket: WebSocket) (context: HttpContext) =
  socket {
    let mutable loop = true

    while loop do
      let! msg = webSocket.read()

      match msg with
      | (Text, data, true) ->
          let reply =
            try
              let json = Encoding.UTF8.GetString data.Span
              let message = JsonSerializer.Deserialize<Message>(json)
              let response = { message with timestamp = System.DateTime.UtcNow }
              Some (JsonSerializer.Serialize(response))
            with _ -> None

          match reply with
          | Some json -> do! webSocket.send Text (Encoding.UTF8.GetBytes json |> ByteSegment) true
          | None -> ()

      | (Ping, data, _) ->
          do! webSocket.send Pong data true

      | (Close, _, _) ->
          do! webSocket.send Close (ByteSegment [||]) true
          loop <- false

      | _ -> ()
  }

let app =
  choose [
    path "/json-ws" >=> checkOrigin >=> handShake jsonWebSocket
  ]

Binary Data Transfer

Send and receive binary data over WebSocket:

open Suave
open Suave.WebSocket
open Suave.Filters
open Suave.Operators
open Suave.Sockets
open Suave.Sockets.Control

let binaryEcho (webSocket: WebSocket) (context: HttpContext) =
  socket {
    let mutable loop = true

    while loop do
      let! msg = webSocket.read()

      match msg with
      | (Binary, data, true) ->
          // Echo binary data back
          do! webSocket.send Binary data true

      | (Ping, data, _) ->
          do! webSocket.send Pong data true

      | (Close, _, _) ->
          do! webSocket.send Close (ByteSegment [||]) true
          loop <- false

      | _ -> ()
  }

let app =
  choose [
    path "/binary" >=> checkOrigin >=> handShake binaryEcho
  ]

Client-Side WebSocket Example

HTML/JavaScript client for WebSocket communication:

<!DOCTYPE html>
<html>
<body>
  <h1>WebSocket Chat</h1>
  <input type="text" id="message" placeholder="Type a message">
  <button onclick="sendMessage()">Send</button>
  <div id="messages"></div>

  <script>
    const ws = new WebSocket('ws://localhost:8080/chat');

    ws.onopen = () => {
      console.log('Connected');
    };

    ws.onmessage = (event) => {
      const div = document.createElement('div');
      div.textContent = event.data;
      document.getElementById('messages').appendChild(div);
    };

    ws.onerror = (error) => {
      console.error('WebSocket error:', error);
    };

    ws.onclose = () => {
      console.log('Disconnected');
    };

    function sendMessage() {
      const input = document.getElementById('message');
      ws.send(input.value);
      input.value = '';
    }
  </script>
</body>
</html>

Error Handling

A failed read or send (peer gone, oversized frame, ...) ends the socket block with an error, which closes the connection. Handle errors in your own message processing where it happens; an exception that escapes the handler closes the socket with code 1011, as described above.

open Suave
open Suave.WebSocket
open Suave.Filters
open Suave.Operators
open Suave.Sockets
open Suave.Sockets.Control

let safeWebSocket (webSocket: WebSocket) (context: HttpContext) =
  socket {
    let mutable loop = true

    while loop do
      // A failed read (peer gone, oversized frame, ...) ends the socket block here
      let! msg = webSocket.read()

      match msg with
      | (Text, data, true) ->
          let reply =
            try
              let str = System.Text.Encoding.UTF8.GetString data.Span
              sprintf "OK: %s" str
            with ex ->
              sprintf "Error: %s" ex.Message
          do! webSocket.send Text (reply |> System.Text.Encoding.UTF8.GetBytes |> ByteSegment) true

      | (Ping, data, _) ->
          do! webSocket.send Pong data true

      | (Close, _, _) ->
          do! webSocket.send Close (ByteSegment [||]) true
          loop <- false

      | _ -> ()
  }

let app =
  choose [
    path "/safe-ws" >=> checkOrigin >=> handShake safeWebSocket
  ]
