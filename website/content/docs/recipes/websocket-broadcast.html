Recipe: Broadcast to WebSocket clients

Goal: Fan-out a message to every connected socket.

Suave does not include a built-in bus. Keep the connected sockets in a MailboxProcessor hub: it handles joins, leaves and broadcasts one at a time, checks the result of every send, and drops clients whose send fails.

open System
open System.Text
open Suave
open Suave.Filters
open Suave.Operators
open Suave.RequestErrors
open Suave.WebSocket
open Suave.Sockets
open Suave.Sockets.Control

/// Browsers send cookies with cross-site WebSocket handshakes and CORS does
/// not apply to them, so only accept pages you serve.
let allowedOrigins = set [ "http://localhost:8080" ]

let checkOrigin : WebPart =
  fun ctx ->
    match ctx.request.header "origin" with
    | Choice1Of2 origin when allowedOrigins.Contains origin -> succeed ctx
    | _ -> FORBIDDEN "Origin not allowed" ctx

type HubMessage =
  | Join of Guid * WebSocket
  | Leave of Guid
  | Broadcast of string

/// Sends one frame; any failure (including an exception) counts as a dead client.
let trySend (ws : WebSocket) (data : ByteSegment) = async {
  try
    let! result = ws.send Text data true |> fun op -> op.AsTask() |> Async.AwaitTask
    return Result.isOk result
  with _ ->
    return false }

/// The hub owns the client table, so joins, leaves and broadcasts are
/// handled one at a time and need no locking.
let hub =
  MailboxProcessor<HubMessage>.Start(fun inbox ->
    let rec loop (clients : Map<Guid, WebSocket>) = async {
      match! inbox.Receive() with
      | Join (id, ws) -> return! loop (Map.add id ws clients)
      | Leave id -> return! loop (Map.remove id clients)
      | Broadcast text ->
          let data = ByteSegment (Encoding.UTF8.GetBytes text)
          let! sent =
            clients
            |> Seq.map (fun (KeyValue (id, ws)) -> async {
                 let! ok = trySend ws data
                 return id, ok })
            |> Async.Parallel
          // Drop every client whose send failed
          let alive =
            sent |> Array.fold (fun m (id, ok) -> if ok then m else Map.remove id m) clients
          return! loop alive }
    loop Map.empty)

let ws (webSocket : WebSocket) (_ : HttpContext) =
  let id = Guid.NewGuid()
  hub.Post (Join (id, webSocket))
  socket {
    try
      let mutable loop = true
      while loop do
        let! msg = webSocket.read()
        match msg with
        | (Text, data, true) ->
            hub.Post (Broadcast (Encoding.UTF8.GetString data.Span))
        | (Ping, data, _) ->
            do! webSocket.send Pong data true
        | (Close, _, _) ->
            do! webSocket.send Close (ByteSegment [||]) true
            loop <- false
        | _ -> ()
    finally
      hub.Post (Leave id)
  }

let app = path "/ws" >=> checkOrigin >=> handShake ws
startWebServer defaultConfig app

Notes:

  • checkOrigin runs before handShake: browsers attach cookies to cross-site WebSocket handshakes and CORS does not apply, so without it any page can open a socket as your user.
  • The handler answers Ping with Pong and the client's Close with Close.
  • A broadcast waits for every send, so one slow client delays the next message for everyone. If that matters, give each client its own queue.
